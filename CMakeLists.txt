cmake_minimum_required(VERSION 3.16)
project(InteractiveFlashcardExaminationEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(flashcard src/main.cpp)
if(MSVC)
  target_compile_options(flashcard PRIVATE /W4)
else()
  target_compile_options(flashcard PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME core_self_test COMMAND flashcard --self-test)
